# Allows you to create your own dynamic keys
# to be called in localization.
# defined_text -> this is it, we're defining the text
# text -> a discrete entry that can be picked to display in loc.
# trigger -> determines if a text entry will be picked or not.
# (The triggers need to be valid for the scope the key is called in
# (eg Root or From.From).)
# localization_key -> points to the localization key
# that'll be used if trigger passes
# 2 - лояльность, 1 - дружелюбие, 0 - безраличность, -1 - враждебность и -2 - непоследовательные

defined_text = {
	name = GetSocialDemocratLoyaltyEffects
	text = {
		trigger = {
			original_tag = UKR
			check_variable = { var = UKR_socialdemocrat_loyalty value = 2 compare = equals }
		}
		localization_key = UKR_faction_loyal_effects
	}
	text = {
		trigger = {
			original_tag = UKR
			check_variable = { var = UKR_socialdemocrat_loyalty value = 1 compare = equals }
		}
		localization_key = UKR_faction_friendly_effects
	}
	text = {
		trigger = {
			original_tag = UKR
			OR =  {	
				check_variable = { var = UKR_socialdemocrat_loyalty value = 0 compare = equals }
				check_variable = { var = UKR_socialdemocrat_loyalty value = -2 compare = equals }
			}
		}
		localization_key = UKR_faction_indifferent_effects
	}
	text = {
		trigger = {
			original_tag = UKR
			check_variable = { var = UKR_socialdemocrat_loyalty value = -1 compare = equals }
		}
		localization_key = UKR_faction_hostile_effects
	}
}
